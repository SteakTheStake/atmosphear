cmake_minimum_required(VERSION 3.20)
project(atmo_audio LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build_profile.json" CACHE PATH "" FORCE)
set(GODOTCPP_PRECISION single CACHE STRING "" FORCE)
set(GODOTCPP_TARGET template_release CACHE STRING "" FORCE)
add_subdirectory("${ATMO_DEPS}/godot-cpp" godot-cpp)
add_library(atmo_audio SHARED src/atmo_audio.cpp)
target_include_directories(atmo_audio PRIVATE "${ATMO_DEPS}/steamaudio/include")
target_link_libraries(atmo_audio PRIVATE godot-cpp "${ATMO_DEPS}/steamaudio/lib/windows-x64/phonon.lib")
target_compile_definitions(atmo_audio PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
  target_compile_options(atmo_audio PRIVATE /O2 /fp:precise /W4)
endif()
set_target_properties(atmo_audio PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/bin")
